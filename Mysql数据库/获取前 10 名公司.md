### 步骤 1：创建测试表（模拟订单表）









```sql
-- 创建订单表（表名：order_table，字段按需求定义）
CREATE TABLE IF NOT EXISTS order_table (
  id INT PRIMARY KEY AUTO_INCREMENT,  -- 主键id
  order_name VARCHAR(100) NOT NULL,   -- 订单名称
  create_time DATETIME NOT NULL,      -- 订单创建时间
  is_valid TINYINT NOT NULL,          -- 订单是否有效（1=有效，0=无效）
  company_name VARCHAR(50) NOT NULL   -- 订单所属公司
);
```

### 步骤 3：执行统计查询（获取前 10 名公司）






```sql
-- 统计各公司订单总数，取前10名
SELECT 
  company_name AS 公司名称,
  COUNT(id) AS 订单数量
FROM 
  order_table
-- 如需统计有效订单，可添加：WHERE is_valid = 1
GROUP BY 
  company_name
ORDER BY 
  订单数量 DESC
LIMIT 10;
```
